       Also add `const router = useRouter();` import from `next/navigation`.

    2. Replace the `handleComplete` stub:
       ```ts
       const handleComplete = async (res: { victory: boolean; battery: number }) => {
         if (res.victory) {
           try {
             await api.completeLevel(ident);
           } catch {
             // level already completed or network error — still show overlay
           }
         }
         setResult({ victory: res.victory });
       };
       ```
       Key constraints:
       - `api.completeLevel` is only called on victory, not defeat
       - Wrap in try/catch so a network error doesn't leave the user stuck on the canvas

    3. Import ResultOverlay dynamically (ssr:false is not needed here — it's pure React, no window access):
       ```ts
       import ResultOverlay from '@/components/game/ResultOverlay';
       ```

    4. In the JSX return, render the overlay conditionally above the `<PhaserGameDynamic>`:
       ```tsx
       return (
         <main>
           <PhaserGameDynamic
             difficulty={airport.difficulty}
             level={airport.level}
             battery={battery}
             onComplete={handleComplete}
           />
           {result !== null && (
             <ResultOverlay
               victory={result.victory}
               airportName={airport.name}
               earnedMoney={airport.speaker_fee}
               onBack={() => router.push('/map')}
               onRetry={() => window.location.reload()}
             />
           )}
         </main>
       );
       ```

    The overlay sits above the canvas via z-index:50 on ResultOverlay — the Phaser canvas has no z-index set, so default (0) applies.
